cmake_minimum_required(VERSION 3.20)
# Generated projects, executables and caches stay outside this directory.
file(RELATIVE_PATH _build_relative "${CMAKE_CURRENT_SOURCE_DIR}" "${CMAKE_BINARY_DIR}")
if(NOT _build_relative MATCHES "^\\.\\./" AND NOT IS_ABSOLUTE "${_build_relative}")
  message(FATAL_ERROR "Use an external build directory: cmake -S <source> -B <separate-directory>")
endif()
project(Car VERSION 1.0 LANGUAGES CXX)
if(NOT WIN32 OR NOT MSVC)
  message(FATAL_ERROR "Car requires Windows and an MSVC toolchain (VS2015 or newer).")
endif()
if(MSVC_VERSION LESS 1900 OR NOT CMAKE_SIZEOF_VOID_P EQUAL 8)
  message(FATAL_ERROR "Use an x64 MSVC toolchain, v140 or newer; runtime dependencies are x64.")
endif()
set(CAR_DEPS_ROOT "${CMAKE_CURRENT_SOURCE_DIR}" CACHE PATH "Directory containing third_party SDK packages")
get_filename_component(CAR_DEPS_ROOT "${CAR_DEPS_ROOT}" ABSOLUTE)
if(NOT DEFINED OpenCV_DIR AND EXISTS "${CAR_DEPS_ROOT}/third_party/opencv/build/x64/vc14/lib/OpenCVConfig.cmake")
  set(OpenCV_DIR "${CAR_DEPS_ROOT}/third_party/opencv/build/x64/vc14/lib" CACHE PATH "OpenCV SDK configuration")
endif()
find_package(OpenCV 4 REQUIRED COMPONENTS core imgproc imgcodecs videoio highgui dnn)
find_path(ONNXRUNTIME_INCLUDE_DIR onnxruntime_c_api.h
  PATHS "${CAR_DEPS_ROOT}/third_party/onnxruntime/build/native/include" NO_DEFAULT_PATH)
if(NOT ONNXRUNTIME_INCLUDE_DIR)
  message(FATAL_ERROR "ONNX Runtime headers missing. Run tools/setup.ps1.")
endif()
add_executable(Car
  src/Car.cpp src/TrackingReview.cpp src/Lighting.cpp src/Text.cpp
  src/Multilingual.cpp src/Inference.cpp src/main.cpp src/TestVideo.cpp src/Tests.cpp)
target_include_directories(Car PRIVATE include "${ONNXRUNTIME_INCLUDE_DIR}" ${OpenCV_INCLUDE_DIRS})
target_compile_features(Car PRIVATE cxx_std_11)
target_compile_definitions(Car PRIVATE NOMINMAX _CRT_SECURE_NO_WARNINGS UNICODE _UNICODE)
target_compile_options(Car PRIVATE /W4 /EHsc /MP /sdl)
target_link_libraries(Car PRIVATE ${OpenCV_LIBS} dxgi gdi32 user32)
set_target_properties(Car PROPERTIES RUNTIME_OUTPUT_DIRECTORY "${CMAKE_BINARY_DIR}/bin/$<CONFIG>")

set(CAR_OPENCV_RUNTIME_DIR "" CACHE PATH "OpenCV DLL directory; auto-detected for the official Windows SDK")
if(NOT CAR_OPENCV_RUNTIME_DIR)
  foreach(_candidate "${OpenCV_DIR}/../bin" "${OpenCV_DIR}/x64/vc14/bin" "${OpenCV_DIR}/x64/vc15/bin")
    if(EXISTS "${_candidate}/opencv_world${OpenCV_VERSION_MAJOR}${OpenCV_VERSION_MINOR}${OpenCV_VERSION_PATCH}.dll")
      set(CAR_OPENCV_RUNTIME_DIR "${_candidate}")
      break()
    endif()
  endforeach()
endif()
file(GLOB _opencv_dlls "${CAR_OPENCV_RUNTIME_DIR}/opencv*.dll")
if(NOT _opencv_dlls)
  message(FATAL_ERROR "Set CAR_OPENCV_RUNTIME_DIR to the matching x64 OpenCV DLL directory.")
endif()
file(GLOB _ocr_dlls "${CAR_DEPS_ROOT}/third_party/tesseract/x64/*.dll")
if(NOT _ocr_dlls)
  message(FATAL_ERROR "Tesseract runtime DLLs missing; run tools/setup.ps1.")
endif()
set(_runtime_dlls ${_opencv_dlls} ${_ocr_dlls}
  "${CAR_DEPS_ROOT}/third_party/onnxruntime/runtimes/win-x64/native/onnxruntime.dll"
  "${CAR_DEPS_ROOT}/third_party/directml/bin/x64-win/DirectML.dll")
foreach(_dll IN LISTS _runtime_dlls)
  if(NOT EXISTS "${_dll}")
    message(FATAL_ERROR "Runtime dependency missing: ${_dll}")
  endif()
  add_custom_command(TARGET Car POST_BUILD COMMAND "${CMAKE_COMMAND}" -E copy_if_different
    "${_dll}" "$<TARGET_FILE_DIR:Car>" VERBATIM)
endforeach()
enable_testing()
if(EXISTS "${CMAKE_CURRENT_SOURCE_DIR}/models/yolox.onnx")
  set(_default_models "${CMAKE_CURRENT_SOURCE_DIR}/models")
else()
  set(_default_models "${CAR_DEPS_ROOT}/models")
endif()
set(CAR_MODELS_DIR "${_default_models}" CACHE PATH "Directory containing ONNX models and tessdata")
add_test(NAME CarSelfTest COMMAND Car --self-test --models "${CAR_MODELS_DIR}")
set_tests_properties(CarSelfTest PROPERTIES WORKING_DIRECTORY "${CMAKE_BINARY_DIR}" TIMEOUT 180)
