cmake_minimum_required(VERSION 3.15)
get_filename_component(sleep_source "${CMAKE_SOURCE_DIR}" REALPATH)
get_filename_component(sleep_binary "${CMAKE_BINARY_DIR}" REALPATH)
string(TOLOWER "${sleep_source}/" sleep_source_lower)
string(TOLOWER "${sleep_binary}/" sleep_binary_lower)
string(FIND "${sleep_binary_lower}" "${sleep_source_lower}" inside_source)
if(inside_source EQUAL 0)
  message(FATAL_ERROR "Build outside the source tree: -B Z:/sleep-build/v142-x64")
endif()
project(Sleep LANGUAGES CXX)
message(STATUS "Toolchain: ${CMAKE_TOOLCHAIN_FILE}")
if(NOT WIN32 OR NOT MSVC)
  message(FATAL_ERROR "Windows/MSVC is required.")
endif()
if(NOT CMAKE_SIZEOF_VOID_P EQUAL 8)
  message(FATAL_ERROR "The supplied neural runtime requires Windows x64.")
endif()
get_filename_component(sleep_parent "${CMAKE_CURRENT_SOURCE_DIR}" DIRECTORY)
set(SLEEP_DATA_ROOT "${sleep_parent}/sleep-data" CACHE PATH "External assets, fixtures and writable results directory")
set(SLEEP_ORT_ROOT "${SLEEP_DATA_ROOT}/third_party/onnxruntime" CACHE PATH "External neural runtime SDK directory")
foreach(external_path SLEEP_DATA_ROOT SLEEP_ORT_ROOT)
  get_filename_component(external_absolute "${${external_path}}" ABSOLUTE)
  file(TO_CMAKE_PATH "${external_absolute}" external_absolute)
  string(TOLOWER "${external_absolute}/" external_lower)
  string(FIND "${external_lower}" "${sleep_source_lower}" inside_source)
  if(inside_source EQUAL 0)
    message(FATAL_ERROR "${external_path} must be outside the source directory")
  endif()
endforeach()
file(TO_CMAKE_PATH "${SLEEP_DATA_ROOT}" SLEEP_DATA_ROOT)
file(MAKE_DIRECTORY "${SLEEP_DATA_ROOT}/tests" "${SLEEP_DATA_ROOT}/output")
add_executable(Sleep src/Model.cpp src/Temporal.cpp src/Image.cpp src/WindowsMedia.cpp src/Vision.cpp src/Application.cpp src/Tests.cpp src/main.cpp)
target_compile_features(Sleep PRIVATE cxx_std_14)
set_target_properties(Sleep PROPERTIES MSVC_RUNTIME_LIBRARY MultiThreadedDLL RUNTIME_OUTPUT_DIRECTORY "${CMAKE_BINARY_DIR}/bin")
target_include_directories(Sleep PRIVATE include "${SLEEP_ORT_ROOT}/include")
target_compile_definitions(Sleep PRIVATE NOMINMAX _CRT_SECURE_NO_WARNINGS SLEEP_DATA_ROOT="${SLEEP_DATA_ROOT}/" WINVER=0x0602 _WIN32_WINNT=0x0602)
target_compile_options(Sleep PRIVATE /W4 /EHsc)
target_link_libraries(Sleep PRIVATE user32 gdi32 ole32 oleaut32 windowscodecs mfplat mfreadwrite mf mfuuid)
foreach(dll onnxruntime.dll DirectML.dll)
  if(NOT EXISTS "${SLEEP_ORT_ROOT}/${dll}")
    message(FATAL_ERROR "Missing neural runtime: ${SLEEP_ORT_ROOT}/${dll}")
  endif()
  add_custom_command(TARGET Sleep POST_BUILD COMMAND ${CMAKE_COMMAND} -E copy_if_different "${SLEEP_ORT_ROOT}/${dll}" "$<TARGET_FILE_DIR:Sleep>" VERBATIM)
endforeach()
add_executable(compiler_probe cmake/compiler_probe.cpp)
set_target_properties(compiler_probe PROPERTIES MSVC_RUNTIME_LIBRARY MultiThreadedDLL RUNTIME_OUTPUT_DIRECTORY "${CMAKE_BINARY_DIR}/bin")
enable_testing()
add_test(NAME sleep_functional COMMAND "$<TARGET_FILE:Sleep>" --self-test)
add_test(NAME compiler_architecture COMMAND "$<TARGET_FILE:compiler_probe>")
file(GENERATE OUTPUT "${CMAKE_BINARY_DIR}/compiler-check.txt" CONTENT "compiler=${CMAKE_CXX_COMPILER}\nversion=${CMAKE_CXX_COMPILER_VERSION}\nhost=${CMAKE_HOST_SYSTEM_NAME}/${CMAKE_HOST_SYSTEM_PROCESSOR}\ntarget=${CMAKE_SYSTEM_NAME}/${CMAKE_SYSTEM_PROCESSOR}\npointer_bytes=${CMAKE_SIZEOF_VOID_P}\ncross_compiling=${CMAKE_CROSSCOMPILING}\n")

install(TARGETS Sleep RUNTIME DESTINATION bin)
install(FILES "${SLEEP_ORT_ROOT}/onnxruntime.dll" "${SLEEP_ORT_ROOT}/DirectML.dll" DESTINATION bin)
install(DIRECTORY "${SLEEP_DATA_ROOT}/models/" DESTINATION data/models FILES_MATCHING PATTERN "*.onnx")
install(FILES "${SLEEP_DATA_ROOT}/testdata/face.png" DESTINATION data/testdata)
install(FILES "${SLEEP_DATA_ROOT}/third_party/onnxruntime_LICENSE" "${SLEEP_DATA_ROOT}/third_party/DirectML_LICENSE.txt" "${SLEEP_DATA_ROOT}/third_party/reference_LICENSE" DESTINATION licenses)
install(DIRECTORY DESTINATION data/tests)
install(DIRECTORY DESTINATION data/output)
